## Приложение: обёртка приёмщика `run_midnight.py`

Запуск из `~/avia-scale/review`:
`PYTHONPATH=src:. $PY run_midnight.py --scenario C --users 100 --phones 2 --tag _rvmidnight`.
Код бота и симулятора не меняет, только подменяет `VirtualClock.time` и дописывает в JSON блок `rv`.

```python
import sys
from datetime import datetime, time as dtime
from loadtest import run as R
from loadtest import vclock as V

DAY = 86400
LIMIT_TEXT = "лимит поисков исчерпан"
_real = V._real_time
SHIFT = datetime.combine(R.TODAY, dtime.min).timestamp() - _real()
V.VirtualClock.time = lambda self: _real() + SHIFT + self.offset
_orig_report = R._report

def _report(args, records, searches, hub, tg, logc, bot, pipeline, t_start, *rest):
    out = _orig_report(args, records, searches, hub, tg, logc, bot, pipeline, t_start, *rest)
    refused = [r for r in records if r.outcome == "refused"]
    limit = [r for r in refused if LIMIT_TEXT in r.text_head]
    out["rv"] = {
        "shift_h": round(SHIFT / 3600, 2),
        "refused_limit": len(limit),
        "refused_other": len(refused) - len(limit),
        "refused_limit_after_24h": sum(1 for r in limit if r.t_click - t_start >= DAY),
        "clicks_total": len(records),
        "clicks_after_24h": sum(1 for r in records if r.t_click - t_start >= DAY),
        "heavy_clicks_after_24h": sum(1 for r in records
                                      if r.t_click - t_start >= DAY and (r.chat_id - 100_000) % 10 == 0),
    }
    return out

R._report = _report
sys.argv = ["loadtest.run", *sys.argv[1:]]
R.main()
```
